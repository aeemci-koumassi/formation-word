Script SQL pour Supabase

Si vous ne l'avez pas encore fait, copiez ce script et collez-le dans le SQL Editor de votre projet Supabase (dashboard.supabase.com) pour créer la table officielle : 

CREATE TABLE IF NOT EXISTS public.recensement_emsp (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    matricule TEXT UNIQUE NOT NULL,
    nom TEXT NOT NULL,
    genre TEXT NOT NULL,
    nationalite TEXT DEFAULT 'Côte d''Ivoire',
    date_naissance DATE,
    whatsapp TEXT NOT NULL,
    telephone_urgence TEXT,
    email TEXT,
    statut TEXT NOT NULL,
    cycle TEXT NOT NULL,
    filiere TEXT NOT NULL,
    niveau TEXT NOT NULL,
    residence TEXT NOT NULL,
    precision_residence TEXT,
    commissions TEXT[],
    competences TEXT,
    groupe_sanguin TEXT,
    remarques TEXT,
    statut_validation TEXT DEFAULT 'Validé'
);

ALTER TABLE public.recensement_emsp ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Lecture et insertion publiques" ON public.recensement_emsp FOR ALL TO public USING (true) WITH CHECK (true);
